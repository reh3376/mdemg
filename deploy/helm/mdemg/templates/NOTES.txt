1. Get the application URL by running these commands:
{{- if .Values.ingress.enabled }}
{{- range $host := .Values.ingress.hosts }}
  {{- range .paths }}
  http{{ if $.Values.ingress.tls }}s{{ end }}://{{ $host.host }}{{ .path }}
  {{- end }}
{{- end }}
{{- else if contains "NodePort" .Values.service.type }}
  export NODE_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "mdemg.fullname" . }})
  export NODE_IP=$(kubectl get nodes --namespace {{ .Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")
  echo http://$NODE_IP:$NODE_PORT
{{- else if contains "LoadBalancer" .Values.service.type }}
     NOTE: It may take a few minutes for the LoadBalancer IP to be available.
           You can watch the status of by running 'kubectl get --namespace {{ .Release.Namespace }} svc -w {{ include "mdemg.fullname" . }}'
  export SERVICE_IP=$(kubectl get svc --namespace {{ .Release.Namespace }} {{ include "mdemg.fullname" . }} --template "{{"{{ range (index .status.loadBalancer.ingress 0) }}{{.}}{{ end }}"}}")
  echo http://$SERVICE_IP:{{ .Values.service.port }}
{{- else if contains "ClusterIP" .Values.service.type }}
  export POD_NAME=$(kubectl get pods --namespace {{ .Release.Namespace }} -l "app.kubernetes.io/name={{ include "mdemg.name" . }},app.kubernetes.io/instance={{ .Release.Name }}" -o jsonpath="{.items[0].metadata.name}")
  export CONTAINER_PORT=$(kubectl get pod --namespace {{ .Release.Namespace }} $POD_NAME -o jsonpath="{.spec.containers[0].ports[0].containerPort}")
  echo "Visit http://127.0.0.1:8080 to use your application"
  kubectl --namespace {{ .Release.Namespace }} port-forward $POD_NAME 8080:$CONTAINER_PORT
{{- end }}

2. Check the health status:
  kubectl --namespace {{ .Release.Namespace }} exec -it deploy/{{ include "mdemg.fullname" . }} -- wget -qO- http://localhost:8080/healthz

3. View Prometheus metrics:
  kubectl --namespace {{ .Release.Namespace }} port-forward svc/{{ include "mdemg.fullname" . }} 8080:{{ .Values.service.port }}
  curl http://localhost:8080/v1/prometheus

{{- if not .Values.secrets.create }}

4. IMPORTANT: You need to create a secret named "{{ .Values.secrets.name }}" with the following keys:
   - neo4j-uri
   - neo4j-user
   - neo4j-pass
   - openai-api-key
   - api-keys

   Example:
   kubectl create secret generic {{ .Values.secrets.name }} \
     --from-literal=neo4j-uri=bolt://neo4j:7687 \
     --from-literal=neo4j-user=neo4j \
     --from-literal=neo4j-pass=YOUR_PASSWORD \
     --from-literal=openai-api-key=YOUR_API_KEY \
     --from-literal=api-keys=YOUR_API_KEYS
{{- end }}
